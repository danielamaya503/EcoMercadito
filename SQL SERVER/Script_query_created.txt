-- ============================================================================
-- TABLAS DE CATÁLOGO
-- ============================================================================

-- Tabla: Departamento (14 departamentos de El Salvador)
CREATE TABLE [Departamento] (
    [DepartamentoId] INT IDENTITY(1,1) NOT NULL,
    [Nombre] NVARCHAR(100) NOT NULL,
    CONSTRAINT [PK_Departamento] PRIMARY KEY CLUSTERED ([DepartamentoId]),
    CONSTRAINT [UQ_Departamento_Nombre] UNIQUE NONCLUSTERED ([Nombre])
);
GO

-- Tabla: Municipio (262 municipios, depende de Departamento)
CREATE TABLE [Municipio] (
    [MunicipioId] INT IDENTITY(1,1) NOT NULL,
    [DepartamentoId] INT NOT NULL,
    [Nombre] NVARCHAR(150) NOT NULL,
    CONSTRAINT [PK_Municipio] PRIMARY KEY CLUSTERED ([MunicipioId]),
    CONSTRAINT [FK_Municipio_Departamento] FOREIGN KEY ([DepartamentoId]) 
        REFERENCES [Departamento]([DepartamentoId]) ON DELETE NO ACTION,
    CONSTRAINT [UQ_Municipio_Nombre_Por_Depto] UNIQUE NONCLUSTERED ([DepartamentoId], [Nombre])
);
GO

-- Tabla: Rol (Comprador, Comercio, Administrador)
CREATE TABLE [Rol] (
    [RolId] INT IDENTITY(1,1) NOT NULL,
    [NombreRol] NVARCHAR(50) NOT NULL,
    CONSTRAINT [PK_Rol] PRIMARY KEY CLUSTERED ([RolId]),
    CONSTRAINT [UQ_Rol_NombreRol] UNIQUE NONCLUSTERED ([NombreRol])
);
GO

-- Tabla: Categoria (catálogo de categorías de productos)
CREATE TABLE [Categoria] (
    [CategoriaId] INT IDENTITY(1,1) NOT NULL,
    [NombreCategoria] NVARCHAR(100) NOT NULL,
    [IconoClase] NVARCHAR(100) NULL,
    [Estado] BIT NOT NULL DEFAULT 1,
    CONSTRAINT [PK_Categoria] PRIMARY KEY CLUSTERED ([CategoriaId]),
    CONSTRAINT [UQ_Categoria_NombreCategoria] UNIQUE NONCLUSTERED ([NombreCategoria])
);
GO

-- ============================================================================
-- TABLAS PRINCIPALES
-- ============================================================================

-- Tabla: Usuario (identidad y autenticación)
CREATE TABLE [Usuario] (
    [UsuarioId] INT IDENTITY(1,1) NOT NULL,
    [Nombre] NVARCHAR(150) NOT NULL,
    [Email] NVARCHAR(255) NOT NULL,
    [Telefono] NVARCHAR(20) NULL,
    [MunicipioId] INT NOT NULL,
    [RolId] INT NOT NULL,
    [Estado] NVARCHAR(20) NOT NULL DEFAULT N'Activo',
    [FechaRegistro] DATETIME2(0) NOT NULL DEFAULT SYSDATETIME(),
    [FechaBaneo] DATETIME2(0) NULL,
    CONSTRAINT [PK_Usuario] PRIMARY KEY CLUSTERED ([UsuarioId]),
    CONSTRAINT [FK_Usuario_Municipio] FOREIGN KEY ([MunicipioId]) 
        REFERENCES [Municipio]([MunicipioId]) ON DELETE NO ACTION,
    CONSTRAINT [FK_Usuario_Rol] FOREIGN KEY ([RolId]) 
        REFERENCES [Rol]([RolId]) ON DELETE NO ACTION,
    CONSTRAINT [UQ_Usuario_Email] UNIQUE NONCLUSTERED ([Email]),
    CONSTRAINT [CHK_Usuario_Estado] CHECK ([Estado] IN (N'Activo', N'Inactivo', N'Suspendido'))
);
GO

-- Índice non-clustered para búsquedas por email (login)
CREATE NONCLUSTERED INDEX [IX_Usuario_Email] ON [Usuario] ([Email]) INCLUDE ([Nombre], [RolId], [Estado]);
GO

-- Índice para filtrar usuarios por rol y estado
CREATE NONCLUSTERED INDEX [IX_Usuario_Rol_Estado] ON [Usuario] ([RolId], [Estado]) INCLUDE ([UsuarioId], [Nombre], [Email]);
GO

-- Tabla: Credencial (almacena hash BCrypt)
CREATE TABLE [Credencial] (
    [CredencialId] INT IDENTITY(1,1) NOT NULL,
    [UsuarioId] INT NOT NULL,
    [PasswordHash] NVARCHAR(255) NOT NULL,
    [ProveedorAuth] NVARCHAR(20) NOT NULL DEFAULT N'Local',
    [ProveedorUserId] NVARCHAR(255) NULL,
    [FechaActualizacion] DATETIME2(0) NOT NULL DEFAULT SYSDATETIME(),
    CONSTRAINT [PK_Credencial] PRIMARY KEY CLUSTERED ([CredencialId]),
    CONSTRAINT [FK_Credencial_Usuario] FOREIGN KEY ([UsuarioId]) 
        REFERENCES [Usuario]([UsuarioId]) ON DELETE CASCADE,
    CONSTRAINT [UQ_Credencial_UsuarioId] UNIQUE NONCLUSTERED ([UsuarioId]),
    CONSTRAINT [CHK_Credencial_ProveedorAuth] CHECK ([ProveedorAuth] IN (N'Local', N'Google', N'Apple'))
);
GO

-- Tabla: Negocio (datos del comercio)
CREATE TABLE [Negocio] (
    [NegocioId] INT IDENTITY(1,1) NOT NULL,
    [UsuarioId] INT NOT NULL,
    [NombreComercial] NVARCHAR(200) NOT NULL,
    [Descripcion] NVARCHAR(1000) NULL,
    [MunicipioId] INT NOT NULL,
    [Estado] NVARCHAR(20) NOT NULL DEFAULT N'Activo',
    [FechaCreacion] DATETIME2(0) NOT NULL DEFAULT SYSDATETIME(),
    CONSTRAINT [PK_Negocio] PRIMARY KEY CLUSTERED ([NegocioId]),
    CONSTRAINT [FK_Negocio_Usuario] FOREIGN KEY ([UsuarioId]) 
        REFERENCES [Usuario]([UsuarioId]) ON DELETE CASCADE,
    CONSTRAINT [FK_Negocio_Municipio] FOREIGN KEY ([MunicipioId]) 
        REFERENCES [Municipio]([MunicipioId]) ON DELETE NO ACTION,
    CONSTRAINT [UQ_Negocio_UsuarioId] UNIQUE NONCLUSTERED ([UsuarioId]),
    CONSTRAINT [CHK_Negocio_Estado] CHECK ([Estado] IN (N'Activo', N'Inactivo'))
);
GO

-- Índice para búsquedas de negocios por municipio
CREATE NONCLUSTERED INDEX [IX_Negocio_Municipio_Estado] ON [Negocio] ([MunicipioId], [Estado]) INCLUDE ([NegocioId], [NombreComercial]);
GO

-- Tabla: OfertaExcedente (publicaciones de productos)
CREATE TABLE [OfertaExcedente] (
    [OfertaId] INT IDENTITY(1,1) NOT NULL,
    [NegocioId] INT NOT NULL,
    [CategoriaId] INT NOT NULL,
    [Titulo] NVARCHAR(150) NOT NULL,
    [PrecioOriginal] DECIMAL(10,2) NOT NULL,
    [PrecioDescuento] DECIMAL(10,2) NOT NULL,
    [CantidadDisponible] INT NOT NULL,
    [FechaLimite] DATE NOT NULL,
    [Disponible] BIT NOT NULL DEFAULT 1,
    [FechaCreacion] DATETIME2(0) NOT NULL DEFAULT SYSDATETIME(),
    CONSTRAINT [PK_OfertaExcedente] PRIMARY KEY CLUSTERED ([OfertaId]),
    CONSTRAINT [FK_OfertaExcedente_Negocio] FOREIGN KEY ([NegocioId]) 
        REFERENCES [Negocio]([NegocioId]) ON DELETE CASCADE,
    CONSTRAINT [FK_OfertaExcedente_Categoria] FOREIGN KEY ([CategoriaId]) 
        REFERENCES [Categoria]([CategoriaId]) ON DELETE NO ACTION,
    CONSTRAINT [CHK_OfertaExcedente_Precios] CHECK ([PrecioDescuento] <= [PrecioOriginal]),
    CONSTRAINT [CHK_OfertaExcedente_Cantidad] CHECK ([CantidadDisponible] >= 0)
);
GO

-- Índice para filtrar ofertas disponibles por categoría y fecha límite
CREATE NONCLUSTERED INDEX [IX_OfertaExcedente_Disponible_Categoria_Fecha] 
    ON [OfertaExcedente] ([Disponible], [CategoriaId], [FechaLimite]) 
    INCLUDE ([OfertaId], [NegocioId], [Titulo], [PrecioDescuento], [CantidadDisponible]);
GO

-- Índice para búsquedas por negocio
CREATE NONCLUSTERED INDEX [IX_OfertaExcedente_Negocio] ON [OfertaExcedente] ([NegocioId]) INCLUDE ([OfertaId], [Titulo], [Disponible]);
GO

-- Vista: Listado de usuarios con rol y ubicación
CREATE OR ALTER VIEW [vw_Usuario_Completo] AS
SELECT 
    u.[UsuarioId],
    u.[Nombre],
    u.[Email],
    u.[Telefono],
    m.[Nombre] AS [Municipio],
    d.[Nombre] AS [Departamento],
    r.[NombreRol] AS [Rol],
    u.[Estado],
    u.[FechaRegistro]
FROM [Usuario] u
INNER JOIN [Municipio] m ON u.[MunicipioId] = m.[MunicipioId]
INNER JOIN [Departamento] d ON m.[DepartamentoId] = d.[DepartamentoId]
INNER JOIN [Rol] r ON u.[RolId] = r.[RolId];
GO

-- Vista: Ofertas activas con datos completos
CREATE OR ALTER VIEW [vw_Oferta_Completa] AS
SELECT 
    o.[OfertaId],
    o.[Titulo],
    o.[PrecioOriginal],
    o.[PrecioDescuento],
    o.[CantidadDisponible],
    o.[FechaLimite],
    o.[Disponible],
    c.[NombreCategoria],
    n.[NombreComercial],
    us.[Nombre] AS [NombreContacto],
    us.[Telefono] AS [TelefonoContacto],
    m.[Nombre] AS [Municipio],
    d.[Nombre] AS [Departamento]
FROM [OfertaExcedente] o
INNER JOIN [Negocio] n ON o.[NegocioId] = n.[NegocioId]
INNER JOIN [Usuario] us ON n.[UsuarioId] = us.[UsuarioId]
INNER JOIN [Categoria] c ON o.[CategoriaId] = c.[CategoriaId]
INNER JOIN [Municipio] m ON n.[MunicipioId] = m.[MunicipioId]
INNER JOIN [Departamento] d ON m.[DepartamentoId] = d.[DepartamentoId]
WHERE o.[Disponible] = 1 AND o.[FechaLimite] >= CAST(SYSDATETIME() AS DATE);
GO

-- ============================================================================
-- STORED PROCEDURES BÁSICOS
-- ============================================================================

-- SP: Registrar nuevo usuario
CREATE OR ALTER PROCEDURE [sp_Usuario_Insertar]
    @Nombre NVARCHAR(150),
    @Email NVARCHAR(255),
    @Telefono NVARCHAR(20),
    @MunicipioId INT,
    @RolId INT,
    @PasswordHash NVARCHAR(255),
    @UsuarioId INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    
    BEGIN TRY
        BEGIN TRANSACTION;
        
        -- Insertar usuario
        INSERT INTO [Usuario] ([Nombre], [Email], [Telefono], [MunicipioId], [RolId])
        VALUES (@Nombre, @Email, @Telefono, @MunicipioId, @RolId);
        
        SET @UsuarioId = SCOPE_IDENTITY();
        
        -- Insertar credencial
        INSERT INTO [Credencial] ([UsuarioId], [PasswordHash])
        VALUES (@UsuarioId, @PasswordHash);
        
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();
        
        RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH
END
GO

-- SP: Autenticar usuario (login)
CREATE OR ALTER PROCEDURE [sp_Usuario_Autenticar]
    @Email NVARCHAR(255),
    @PasswordHash NVARCHAR(255)
AS
BEGIN
    SET NOCOUNT ON;
    
    SELECT 
        u.[UsuarioId],
        u.[Nombre],
        u.[Email],
        u.[Telefono],
        u.[MunicipioId],
        u.[RolId],
        r.[NombreRol] AS [Rol],
        u.[Estado]
    FROM [Usuario] u
    INNER JOIN [Credencial] c ON u.[UsuarioId] = c.[UsuarioId]
    INNER JOIN [Rol] r ON u.[RolId] = r.[RolId]
    WHERE u.[Email] = @Email 
      AND c.[PasswordHash] = @PasswordHash
      AND u.[Estado] = N'Activo'
      AND c.[ProveedorAuth] = N'Local';
END
GO

-- SP: Insertar nueva oferta
CREATE OR ALTER PROCEDURE [sp_Oferta_Insertar]
    @NegocioId INT,
    @CategoriaId INT,
    @Titulo NVARCHAR(150),
    @PrecioOriginal DECIMAL(10,2),
    @PrecioDescuento DECIMAL(10,2),
    @CantidadDisponible INT,
    @FechaLimite DATE
AS
BEGIN
    SET NOCOUNT ON;
    
    BEGIN TRY
        -- Validar que el negocio exista y esté activo
        IF NOT EXISTS (SELECT 1 FROM [Negocio] WHERE [NegocioId] = @NegocioId AND [Estado] = N'Activo')
        BEGIN
            RAISERROR (N'El negocio no existe o está inactivo.', 16, 1);
            RETURN;
        END
        
        -- Validar que la categoría exista y esté activa
        IF NOT EXISTS (SELECT 1 FROM [Categoria] WHERE [CategoriaId] = @CategoriaId AND [Estado] = 1)
        BEGIN
            RAISERROR (N'La categoría no existe o está inactiva.', 16, 1);
            RETURN;
        END
        
        -- Validar que el precio de descuento no sea mayor al original
        IF @PrecioDescuento > @PrecioOriginal
        BEGIN
            RAISERROR (N'El precio de descuento no puede ser mayor al precio original.', 16, 1);
            RETURN;
        END
        
        -- Validar cantidad
        IF @CantidadDisponible < 0
        BEGIN
            RAISERROR (N'La cantidad disponible no puede ser negativa.', 16, 1);
            RETURN;
        END
        
        INSERT INTO [OfertaExcedente] 
            ([NegocioId], [CategoriaId], [Titulo], [PrecioOriginal], [PrecioDescuento], 
             [CantidadDisponible], [FechaLimite])
        VALUES 
            (@NegocioId, @CategoriaId, @Titulo, @PrecioOriginal, @PrecioDescuento, 
             @CantidadDisponible, @FechaLimite);
    END TRY
    BEGIN CATCH
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();
        
        RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH
END
GO

-- ============================================================================
-- TRIGGER: Actualizar estado de oferta cuando se agota
-- ============================================================================

CREATE OR ALTER TRIGGER [trg_OfertaExcedente_ActualizarDisponible]
ON [OfertaExcedente]
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Si la cantidad disponible llega a 0, marcar como no disponible automáticamente
    UPDATE o
    SET [Disponible] = 0
    FROM [OfertaExcedente] o
    INNER JOIN inserted i ON o.[OfertaId] = i.[OfertaId]
    WHERE i.[CantidadDisponible] = 0 AND i.[Disponible] = 1;
END
GO
